project(idkengine_network)

if(NOT TARGET libidk_remote)
    message(FATAL_ERROR "Engine networking requires IDK_BUILD_REMOTE=ON")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Networking session entropy currently uses Linux getrandom")
endif()
add_library(idkengine_network STATIC src/World.cpp src/Session.cpp)
target_compile_features(idkengine_network PUBLIC cxx_std_23)
target_include_directories(idkengine_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(idkengine_network PUBLIC libidk_remote)
target_compile_options(idkengine_network PRIVATE -Wall -Wextra -Werror -fno-exceptions)

add_executable(idk_net_demo examples/Main.cpp)
target_link_libraries(idk_net_demo PRIVATE idkengine_network)
target_compile_options(idk_net_demo PRIVATE -Wall -Wextra -Werror -fno-exceptions)
install(TARGETS idk_net_demo RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

if(BUILD_TESTING)
    add_executable(idk_network_tests tests/Network.cpp)
    target_link_libraries(idk_network_tests PRIVATE idkengine_network SDL3_net::SDL3_net SDL3::SDL3)
    target_include_directories(idk_network_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_options(idk_network_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions)
    add_test(NAME idk_network_tests COMMAND idk_network_tests)
    set_tests_properties(idk_network_tests PROPERTIES TIMEOUT 30)
endif()
